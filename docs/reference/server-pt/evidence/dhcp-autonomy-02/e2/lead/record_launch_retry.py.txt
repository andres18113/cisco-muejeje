"""Retry only the refused, read-only launch record with a fresh OS title."""

from __future__ import annotations

import json
import os
import subprocess
import sys
from pathlib import Path

from packet_tracer_mcp.application.use_cases.server_pt_process_evidence import (
    incarnation_ticks,
    packet_tracer_process_role,
)
from packet_tracer_mcp.infrastructure.execution.file_bridge import bridge_dir
from packet_tracer_mcp.infrastructure.execution.server_pt_process_control import (
    PowerShellOwnedProcessControl,
)
from packet_tracer_mcp.infrastructure.persistence.campaign_coordination import (
    FileCampaignCoordinator,
)

ROOT = Path(
    r"C:\Users\Andres\Desktop\Universidad\Uce\Cuarto\Infra\Cisco-MCP-server-services-goal-foundations"
)
EVIDENCE = Path(__file__).resolve().parent
ids = json.loads((EVIDENCE / "ids.json").read_text(encoding="utf-8"))
first = json.loads((EVIDENCE / "04-launch-capture.json").read_text(encoding="utf-8"))
prior = json.loads((EVIDENCE / "05-record-launch.json").read_text(encoding="utf-8"))
if prior["exit_code"] != 2:
    raise RuntimeError("prior_launch_refusal_not_established")
coordinator = FileCampaignCoordinator()
control = PowerShellOwnedProcessControl()
claim = coordinator.claim_lifecycle(attempt_id=ids["Attempt"])
try:
    observed = control.observe(first["pid"])
    census = control.packet_tracer_processes()
    pending = sorted(
        item.name
        for item in bridge_dir().iterdir()
        if item.is_file() and item.name != "alive.txt"
    )
    launch_identity = {
        **first,
        "observed_command_line": first["command_line"],
    }
    ticks = incarnation_ticks(first["process_incarnation"])
    roles = [
        packet_tracer_process_role(launch_identity, ticks, row)
        for row in census.processes
    ] if ticks is not None else []
    if (
        observed.error
        or not observed.present
        or observed.process_path != first["process_path"]
        or observed.process_incarnation != first["process_incarnation"]
        or observed.command_line != first["command_line"]
        or not observed.main_window_title
        or census.error
        or roles.count("owned") != 1
        or any(role not in {"owned", "owned_helper"} for role in roles)
        or pending
        or coordinator.verify(claim)
    ):
        raise RuntimeError("fresh_launch_retry_identity_unestablished")
    capture = {**first, "main_window_title": observed.main_window_title}
    with (EVIDENCE / "04b-launch-capture.json").open("x", encoding="utf-8") as out:
        json.dump(capture, out, sort_keys=True, indent=2)
        out.write("\n")
    env = dict(os.environ)
    env["PT_MCP_GOVERNED_ROOT"] = str(ROOT)
    result = subprocess.run(
        [
            sys.executable,
            "-m",
            "packet_tracer_mcp.adapters.cli.server_pt_commissioning",
            "--record-launch",
            "--campaign",
            "dhcp-autonomy",
            "--attempt",
            ids["Attempt"],
            "--charter",
            str(ROOT / "docs/reference/server-pt/assignments/ServerPT_DHCP_Delegated_Autonomy_Mandate.md"),
            "--launch-evidence",
            str(EVIDENCE / "04b-launch-capture.json"),
        ],
        cwd=ROOT,
        env=env,
        capture_output=True,
        text=True,
    )
    with (EVIDENCE / "05b-record-launch.json").open("x", encoding="utf-8") as out:
        json.dump(
            {"exit_code": result.returncode, "stdout": result.stdout, "stderr": result.stderr},
            out,
            sort_keys=True,
            indent=2,
        )
        out.write("\n")
    if result.returncode != 0:
        raise RuntimeError("fresh_launch_retry_refused")
finally:
    with (EVIDENCE / "06b-launch-claim-release.json").open("x", encoding="utf-8") as out:
        json.dump(
            {"claim": claim.compact_summary(), "findings": list(coordinator.release(claim))},
            out,
            sort_keys=True,
            indent=2,
        )
        out.write("\n")
