"""Retire only this episode's recorded disposable Packet Tracer process."""

from __future__ import annotations

import json
import os
import subprocess
import sys
from pathlib import Path

from authority import (
    launch_capture,
    prepared_plan,
    require_opening,
    verify_tools,
)

from packet_tracer_mcp.application.use_cases.server_pt_campaign import SP2_CAMPAIGN
from packet_tracer_mcp.infrastructure.persistence.server_pt_commissioning_store import (
    ServerPtCommissioningStore,
)

ROOT = Path(
    r"C:\Users\Andres\Desktop\Universidad\Uce\Cuarto\Infra\Cisco-MCP-server-services-goal-foundations"
)
EPISODE = Path(sys.argv[1]).resolve()
if not EPISODE.is_relative_to(ROOT / "data" / "services" / "sp2-governed"):
    raise RuntimeError("episode_path_outside_governed_root")
verify_tools(EPISODE, Path(__file__))
ids = json.loads((EPISODE / "ids.json").read_text(encoding="utf-8"))
prepared_plan(EPISODE, ids)
store = ServerPtCommissioningStore(ROOT, SP2_CAMPAIGN.campaign_id)
require_opening(ROOT, EPISODE, ids, store)
launch_capture(EPISODE, ids, store)
# Optional retry suffix ("b", "c"): each maintained --retire invocation keeps its own files.
SUFFIX = sys.argv[2] if len(sys.argv) > 2 else ""
if SUFFIX not in ("", "b", "c"):
    raise RuntimeError("retire_suffix_invalid")
verify_tools(EPISODE, Path(__file__))
require_opening(ROOT, EPISODE, ids, store)
env = dict(os.environ)
env["PT_MCP_GOVERNED_ROOT"] = str(ROOT)
args = [
    sys.executable,
    "-B",
    "-m",
    "packet_tracer_mcp.adapters.cli.server_pt_commissioning",
    "--retire",
    "--campaign",
    "sp2",
    "--attempt",
    ids["attempt_id"],
    "--charter",
    ids["charter"],
]
with (EPISODE / f"10{SUFFIX}-retire.stdout").open("x", encoding="utf-8") as stdout:
    with (EPISODE / f"10{SUFFIX}-retire.stderr").open("x", encoding="utf-8") as stderr:
        result = subprocess.run(
            args, cwd=ROOT, env=env, stdout=stdout, stderr=stderr, timeout=360
        )
(EPISODE / f"10{SUFFIX}-retire.exit").write_text(
    str(result.returncode) + "\n", encoding="utf-8"
)
raise SystemExit(result.returncode)
