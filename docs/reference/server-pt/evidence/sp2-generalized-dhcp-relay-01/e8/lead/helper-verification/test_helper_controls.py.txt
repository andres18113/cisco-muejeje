def save(path: Path, value: object) -> None:
    """Write a fixture record in the actual helper stdout or JSON format."""
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(
        json.dumps(
            value, sort_keys=True, indent=None if path.suffix == ".stdout" else 2
        )
        + "\n"
    )


def digest(path: Path) -> str:
    """Hash independent fixture bytes."""
    return hashlib.sha256(path.read_bytes()).hexdigest()

def execute(lab, name, *, move_change=None, argv=None):
    """Run real helper statements, replacing only external observation/effects."""
    program = ast.parse((lab.tools / name).read_text())
    for node in program.body:
        if isinstance(node, ast.Assign) and any(
            isinstance(t, ast.Name) and t.id in {"ROOT", "EXE"} for t in node.targets
        ):
            node.value = ast.Constant(
                str(
                    lab.root
                    if node.targets[0].id == "ROOT"
                    else lab.root / "PacketTracer.exe"
                )
            )
            node.value = ast.Call(
                func=ast.Name(id="Path", ctx=ast.Load()), args=[node.value], keywords=[]
            )
    ast.fix_missing_locations(program)
    claims = []

    class Coordinator:
        def claim_lifecycle(self, **kwargs):
            claims.append(kwargs)
            raise RuntimeError("offline_effect_boundary_reached")

    class Control:
        def packet_tracer_processes(self):
            return SimpleNamespace(error="", processes=[])

    def run(args, **kwargs):
        if args[0] == "git":
            values = {
                "HEAD": HEAD,
                "HEAD^{tree}": TREE,
                "--porcelain=v1": "",
                "--show-current": "feature/server-pt-goal-foundations",
            }
            return SimpleNamespace(stdout=values.get(args[-1], ""))
        raise AssertionError("external process attempted")

    replace = os.replace

    def moved(a, b):
        replace(a, b)
        if move_change == "bytes":
            (Path(b) / "README.md").write_text("changed after move")
        if move_change == "extra":
            (Path(b) / "extra.txt").write_text("unexpected")

    sys.modules.pop("authority", None)
    sys.modules.pop("publish_seal", None)
    sys.path.insert(0, str(lab.tools))
    try:
        with (
            patch.object(sys, "dont_write_bytecode", True),
            patch.object(
                sys, "argv", argv or [str(lab.tools / name), str(lab.lead), "8"]
            ),
            patch("subprocess.run", run),
            patch("os.replace", moved),
            patch(
                "packet_tracer_mcp.infrastructure.persistence.server_pt_commissioning_store.ServerPtCommissioningStore",
                lab.Store,
            ),
            patch(
                "packet_tracer_mcp.infrastructure.persistence.campaign_coordination.FileCampaignCoordinator",
                Coordinator,
            ),
            patch(
                "packet_tracer_mcp.infrastructure.execution.server_pt_process_control.PowerShellOwnedProcessControl",
                Control,
            ),
            patch(
                "packet_tracer_mcp.infrastructure.execution.file_bridge.bridge_dir",
                lambda: lab.root / "mailbox",
            ),
            patch(
                "packet_tracer_mcp.infrastructure.execution.import_isolation_preflight.ImportIsolationPreflight.ensure_isolated",
                lambda _: SimpleNamespace(
                    isolated=True, state=SimpleNamespace(value="ISOLATED")
                ),
            ),
        ):
            try:
                exec(
                    compile(program, str(lab.tools / name), "exec"),
                    {"__name__": "__main__", "__file__": str(lab.tools / name)},
                )
                return None, claims
            except (RuntimeError, OSError, ValueError, SystemExit) as exc:
                return exc, claims
    finally:
        sys.path.pop(0)
